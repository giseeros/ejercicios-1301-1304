# Creo una función llamada prueba que recibe a y b.
def prueba(a, b):

    # Aumento el valor de a en 1.
    a = a + 1

    # Aumento el valor de b en 2.
    b = b + 2

    # Devuelvo los nuevos valores de a y b.
    return (a, b)
