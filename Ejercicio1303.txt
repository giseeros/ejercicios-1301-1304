CÓDIGO
# Esta función cuenta cuántos números tiene el DNI.
def validar_dni(dni):

    # Comienzo el contador en 0.
    cantidad = 0

    # Mientras el DNI todavía tenga números, sigo contando.
    while dni != 0:

        # Sumo 1 al contador.
        cantidad = cantidad + 1

        # Quito el último número del DNI.
        dni = dni // 10

    # Devuelvo True si tiene exactamente 8 números.
    return cantidad == 8
